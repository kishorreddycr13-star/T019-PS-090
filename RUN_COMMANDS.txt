python scripts/verify_schema.py

--------------------------------------------------------------------------------

python run_engine.py

python run_engine.py sample_data evaluation_outputs

6. OPERATIONAL MANAGEMENT CLI (`management_cli.py`)
--------------------------------------------------------------------------------
# 1. Rights Calendar report for a title:
python cli/management_cli.py report calendar --title-id TTL-001

# 2. Monthly Licensor Settlement Statement:
python cli/management_cli.py report settlement --licensor-id LIC-01 --month 2026-07

# 3. View Decisions Report over a date window:
python cli/management_cli.py report views --from-date 2026-07-01 --to-date 2026-07-01

# 4. Check Playback Entitlement at a specific timestamp:
python cli/management_cli.py playback check --title-id TTL-001 --territory IN --start-utc "2026-07-01 12:00:00"

# 5. View Consumption Metering summary for a month/territory:
python cli/management_cli.py metering summary --month 2026-07 --territory IN

# 6. View Subscription & Ad Revenue Title Shares:
python cli/management_cli.py revenue title-shares --month 2026-07 --territory IN

# 7. Query specific licence contract details:
python cli/management_cli.py licence get --licence-id LCN-001

# 8. List all registered licensors:
python cli/management_cli.py licensor list

# 9. Trigger monthly settlement calculation pipeline:
python cli/management_cli.py settlement run
